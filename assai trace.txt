CREATE TABLE TELESSVR.AUD_REPEMPR002 (
  ID_AUDITORIA    NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  DATA_EVENTO     TIMESTAMP DEFAULT SYSTIMESTAMP NOT NULL,
  OPERACAO        VARCHAR2(10) NOT NULL, -- 'INSERT', 'UPDATE' ou 'DELETE'
  -- O QUE ESTAVA ANTES (:OLD)
  IFUNC_ANTES     CHAR(12),
  PIS_ANTES       CHAR(12),
  NOME_ANTES      VARCHAR2(52),
  CPF_ANTES       CHAR(11),
  -- O QUE FICOU AGORA (:NEW)
  IFUNC_DEPOIS    CHAR(12),
  PIS_DEPOIS      CHAR(12),
  NOME_DEPOIS     VARCHAR2(52),
  CPF_DEPOIS      CHAR(11),
  -- 
  USUARIO_BANCO   VARCHAR2(100),
  USUARIO_SO      VARCHAR2(100),
  MAQUINA_HOST    VARCHAR2(100),
  IP_ORIGEM       VARCHAR2(50),
  PROGRAMA_EXECUT VARCHAR2(200), -- java.exe, httpd.exe, Conex.exe, etc.
  MODULO_APLIC    VARCHAR2(200),
  SQL_EXECUTADO   CLOB,          -- Pega a query exata que rodou!
  CALL_STACK      CLOB           -- Se foi procedure/trigger do banco, mostra a linha exata
);



CREATE OR REPLACE TRIGGER TELESSVR.TRG_AUD_REPEMPR002
AFTER INSERT OR UPDATE OR DELETE ON TELESSVR.REPEMPR002
FOR EACH ROW
DECLARE
  PRAGMA AUTONOMOUS_TRANSACTION;

  v_operacao        VARCHAR2(10);
  v_usuario_banco   VARCHAR2(100);
  v_usuario_so      VARCHAR2(100);
  v_maquina_host    VARCHAR2(100);
  v_ip_origem       VARCHAR2(50);
  v_programa        VARCHAR2(200);
  v_modulo          VARCHAR2(200);
  v_sid             NUMBER;
  v_sql_id          VARCHAR2(50);
  v_sql_texto       CLOB;
  v_call_stack      CLOB;
BEGIN
  IF INSERTING THEN
    v_operacao := 'INSERT';
  ELSIF UPDATING THEN
    v_operacao := 'UPDATE';
  ELSIF DELETING THEN
    v_operacao := 'DELETE';
  END IF;

  v_usuario_banco := SYS_CONTEXT('USERENV', 'SESSION_USER');
  v_usuario_so    := SYS_CONTEXT('USERENV', 'OS_USER');
  v_maquina_host  := SYS_CONTEXT('USERENV', 'HOST');
  v_ip_origem     := SYS_CONTEXT('USERENV', 'IP_ADDRESS');
  v_modulo        := SYS_CONTEXT('USERENV', 'MODULE');
  v_sid           := TO_NUMBER(SYS_CONTEXT('USERENV', 'SID'));
  v_call_stack    := TO_CLOB(DBMS_UTILITY.FORMAT_CALL_STACK);

  -- Busca programa e o SQL_ID exato da sessao
  BEGIN
    EXECUTE IMMEDIATE 'SELECT program, sql_id FROM v$session WHERE sid = :1'
       INTO v_programa, v_sql_id
      USING v_sid;

    IF v_sql_id IS NOT NULL THEN
      EXECUTE IMMEDIATE 'SELECT sql_fulltext FROM v$sql WHERE sql_id = :1 AND ROWNUM = 1'
         INTO v_sql_texto
        USING v_sql_id;
    END IF;
  EXCEPTION
    WHEN OTHERS THEN
      v_programa  := NVL(v_modulo, 'SEM_ACESSO_VSESSION');
      v_sql_texto := 'Sem privilegio na V$SESSION / V$SQL';
  END;

  INSERT INTO telessvr.aud_repempr002 (
    operacao,
    ifunc_antes, pis_antes, nome_antes, cpf_antes,
    ifunc_depois, pis_depois, nome_depois, cpf_depois,
    usuario_banco, usuario_so, maquina_host, ip_origem,
    programa_execut, modulo_aplic, sql_executado, call_stack
  ) VALUES (
    v_operacao,
    :OLD.ifunc, :OLD.pis, :OLD.nome, :OLD.cpf,
    :NEW.ifunc, :NEW.pis, :NEW.nome, :NEW.cpf,
    v_usuario_banco, v_usuario_so, v_maquina_host, v_ip_origem,
    v_programa, v_modulo, v_sql_texto, v_call_stack
  );

  COMMIT;
EXCEPTION
  WHEN OTHERS THEN
    ROLLBACK;
END;
/

GRANT SELECT ON SYS.V_$SESSION TO TELESSVR;
GRANT SELECT ON SYS.V_$SQL TO TELESSVR;
